# AmneziaWG feed. Append to your FEEDS variable in build-local.sh, or add to
# the builder's feeds list:
#
#   FEEDS="src-git nss https://github.com/JuliusBairaktaris/nss-packages.git;edma-nss
#          src-git awg https://github.com/2Grey/awg-openwrt.git"
#
# Provides kmod-amneziawg, amneziawg-tools and luci-proto-amneziawg, all three
# pinned in devices/common/config.
#
# The kmod builds out of tree against the running kernel, so it is one more
# thing that can break on a kernel bump — if a future build fails in
# kmod-amneziawg, dropping this feed and the three CONFIG_PACKAGE lines gets
# you back to a working image, since nothing routes through awg unless
# inject-secrets.sh was run with TUNNEL_PROTO=amneziawg.
